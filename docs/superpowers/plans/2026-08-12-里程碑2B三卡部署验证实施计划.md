# 里程碑 2B 三卡部署验证实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 `192.168.29.11` 上可重复构建并运行四个平台服务、四类基础设施和 24 个算子实例，证明 18 个 GPU 实例真实使用指定 GPU，八类算子均完成一次真实业务调用，并留下可审计、可恢复的里程碑 2B-Deploy 证据。

**Architecture:** 使用 `docker-compose.infrastructure.yml`、`docker-compose.platform.yml` 和 `docker-compose.operators.yml` 三层 Compose；算子层通过 `gpu0`、`gpu1`、`gpu2`、`cpu` profiles 渐进启动。每个 GPU 容器只看到一张物理 GPU，容器内统一使用 `cuda:0`，算子通过 control-service 注册、心跳并领取容量；部署 Harness 只负责环境、容器、GPU、注册、直接推理和恢复证据，不绕过平台写数据库状态。

**Tech Stack:** 平台与除 FaceRec 外的算子使用 Python 3.11，FaceRec 镜像暂时使用 Python 3.10/FastDeploy 1.0.7；其余包括 FastAPI/Uvicorn、Docker 26、Docker Compose 2.27、NVIDIA Container Runtime、PostgreSQL 17、Redis 7.4、Kafka 4.0、MongoDB 7、Pytest、Bash、OpenSpec 和 Git。

```python
from pathlib import Path

import yaml


DEPLOY = Path(__file__).resolve().parents[1] / "deploy"
GPU_OPERATORS = {
    "asr-offline": (8083, 18083),
    "asr-online": (8084, 18084),
    "ocr": (8866, 18866),
    "vbas": (8981, 18981),
    "facerec": (8000, 18003),
    "screen-det": (8880, 18880),
}
CPU_OPERATORS = {
    "ppt-slice": (9001, 19001),
    "text-analysis": (8000, 18000),
}


def expected_gpu_instances() -> dict[str, tuple[str, str]]:
    result = {}
    for gpu in range(3):
        for operator in GPU_OPERATORS:
            name = f"{operator}-gpu{gpu}"
            result[name] = (f"gpu{gpu}", str(gpu))
    return result


def expected_cpu_instances() -> set[str]:
    return {
        f"{operator}-cpu{index}"
        for operator in CPU_OPERATORS
        for index in range(3)
    }


def test_operator_compose_declares_exact_24_instance_topology() -> None:
    document = yaml.safe_load(
        (DEPLOY / "docker-compose.operators.yml").read_text(encoding="utf-8")
    )
    services = document["services"]
    assert set(services) == set(expected_gpu_instances()) | expected_cpu_instances()
    for name, (profile, gpu_id) in expected_gpu_instances().items():
        service = services[name]
        assert service["profiles"] == [profile]
        assert service["environment"]["PLATFORM_GPU_ID"] == gpu_id
        assert service["environment"]["GPU_PROCESS_NAME"] == name.rsplit("-gpu", 1)[0].replace("-", "_")
        assert service["deploy"]["resources"]["reservations"]["devices"] == [{
            "driver": "nvidia",
            "device_ids": [gpu_id],
            "capabilities": ["gpu"],
        }]
    for name in expected_cpu_instances():
        assert services[name]["profiles"] == ["cpu"]
        assert "deploy" not in services[name] or "reservations" not in services[name].get("deploy", {}).get("resources", {})
```

```python
def require_gpu_enabled() -> bool:
    return os.getenv("REQUIRE_GPU", "false").strip().lower() in {"1", "true", "yes"}


def resolve_runtime_device(configured: str) -> torch.device:
    wants_cuda = configured.startswith("cuda:")
    if require_gpu_enabled() and not wants_cuda:
        raise RuntimeError("部署要求使用 GPU，但算子配置不是 cuda:<index>")
    if wants_cuda and not torch.cuda.is_available():
        raise RuntimeError(f"算子要求使用 GPU {configured}，但 CUDA 不可用")
    return torch.device(configured if wants_cuda else "cpu")
```
